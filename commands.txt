MesterX Ultra v6 - A to Z Commands
=================================

A) Open project
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION

B) Install frontend dependencies
cd frontend
npm install

C) Run frontend development server
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION\frontend
npm run dev

D) Build frontend for production
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION\frontend
npm run build

E) Start PostgreSQL only with Docker Compose
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION
docker compose --env-file .env.production up -d db

F) Stop PostgreSQL container
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION
docker compose --env-file .env.production stop db

G) View PostgreSQL container status
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION
docker compose --env-file .env.production ps

H) View PostgreSQL logs
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION
docker compose --env-file .env.production logs db

I) Run backend API locally with environment variables
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION
powershell -NoProfile -Command "$env:ASPNETCORE_ENVIRONMENT='Production'; $env:ASPNETCORE_URLS='http://localhost:5000'; $env:ConnectionStrings__DefaultConnection='Host=localhost;Port=5432;Database=mesterxdb;Username=mesterx;Password=CHANGE_THIS_STRONG_PASSWORD_123!;Include Error Detail=true'; $env:Jwt__Secret='CHANGE_THIS_TO_64_CHAR_MIN_SECRET_KEY_openssl_rand_hex_64_here'; $env:Jwt__Issuer='mesterx-api'; $env:Jwt__Audience='mesterx-client'; $env:Jwt__ExpiryMinutes='60'; $env:Jwt__RefreshTokenExpiryDays='30'; $env:AllowedOrigins__0='http://localhost:3000'; dotnet run --project backend/src/MesterX.API/MesterX.API.csproj"

J) Restore backend packages
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION
dotnet restore backend/src/MesterX.API/MesterX.API.csproj

K) Build backend API
cd c:\projects\MesterX-Ultra-v6-FINAL-PRODUCTION
dotnet build backend/src/MesterX.API/MesterX.API.csproj
